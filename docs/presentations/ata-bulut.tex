\begin{frame}[fragile]{\file{convert.py} mimari}
\begin{lstlisting}
def convert(source=None, out=None):
    src = source if source else _pick_source(prefer_csv=True)
    if src.suffix.lower() == ".csv":
        raw = _read_csv(src)          # 7 sat. preamble atla
    else:
        raw = _read_xlsx(src)         # "Units" satirini drop

    df = _normalize_columns(raw)      # P1 ya da "P1 - Force..."
    df = pd.to_numeric(...).dropna()
    df = _convert_units(df)           # MPa -> Pa, mm -> m
    df = _attach_constants(df, ModelConstants())
    df["case_id"] = [f"DP_{i+1:03d}" for i in range(len(df))]
    df.to_csv(out, index=False)
    return df
\end{lstlisting}

  \begin{itemize}\small
    \item İki kaynak formatı: CSV (preamble + bare \code{P1}) ve XLSX
          (Units satırı + \code{P1 - Force X Component})
    \item Tek regex \verb|^\s*(P\d+)\b| iki formatı da yakalar
    \item Birim normalizasyonu: MPa\(\to\)Pa, mm\(\to\)m
    \item ANSYS sabitleri (mesh, malzeme) \code{ModelConstants}'tan satırlara
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{\file{preprocess.py}: özellik mühendisliği}
\begin{lstlisting}
INPUT_FEATURES = ["patient_mass_kg", "K_factor", "force_angle_deg"]
DERIVED_FEATURES = ["force_x_N", "force_z_N", "force_magnitude_N"]

def add_features(df):
    out = df.copy()
    rad = np.deg2rad(out["force_angle_deg"])
    Fmag = out["patient_mass_kg"] * 9.81 * out["K_factor"]
    out["force_x_N"] = Fmag * np.cos(rad)
    out["force_z_N"] = Fmag * np.sin(rad)
    out["force_magnitude_N"] = Fmag
    return out
\end{lstlisting}

  \vspace{0.2cm}
  \begin{itemize}\small
    \item Model 3 bağımsız girdi görüyor: \(m, K, \theta\)
    \item \(F_x, F_z, |F|\) bu üçünden \textbf{analitik} türev --- predict ile
          train aynı formüle bağlı, drift yok
    \item \(F_y\) ve \code{fixed\_support\_n\_faces} tüm satırlarda sabit
          (varyans = 0) \(\Rightarrow\) girdiden düşürüldü
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{CV gruplandırma --- \code{grid\_groups()}}
\begin{lstlisting}
GRID_KEYS = ("patient_mass_kg", "K_factor", "force_angle_deg")

def grid_groups(df):
    """Return a group ID per row, equal for ANSYS re-runs."""
    grouped = df[list(GRID_KEYS)].astype(str).agg("|".join, axis=1)
    return grouped.to_numpy()
\end{lstlisting}

  \vspace{0.4cm}
  \textbf{Recep'in \code{train.py}'da kullanımı:}
  \begin{itemize}\small
    \item \code{GroupKFold(n\_splits=5).split(X, y, groups=grid\_groups(df))}
    \item Aynı \((m, K, \theta)\) üçlüsü her zaman aynı fold'da
    \item Olmasaydı: \(R^2\) yapay olarak \(\approx 1.0\)'a şişerdi (leakage)
  \end{itemize}
\end{frame}

\begin{frame}{Yol boyu karşılaştığım hatalar}
  \begin{itemize}
    \item \textbf{XLSX'in ``Units'' satırı:} İlk veri satırı aslında birim
          etiketi (\code{kg, MPa, mm, ...}) --- pandas tüm sütunu
          \code{object} dtype yaptı, sayısal işlemler çöktü.
          \alert{Çözüm:} satırı drop + \code{pd.to\_numeric(errors=`coerce')}
    \item \textbf{İki başlık formatı:} CSV'de \code{P1}, XLSX'te
          \code{P1 - Force X Component}. \alert{Çözüm:} regex
          \verb|^\s*(P\d+)\b| ile her ikisini yakalayan tek parser.
    \item \textbf{\code{out=OUT\_CSV} varsayılan argümanı:} Python'da
          fonksiyon tanım zamanında bağlandığı için test'lerde
          \code{monkeypatch} işe yaramıyordu \(\Rightarrow\) \code{data/dataset.csv}
          gerçek dosyaya yazıyordu. \alert{Çözüm:} \code{out=None} +
          runtime resolve.
  \end{itemize}
\end{frame}
