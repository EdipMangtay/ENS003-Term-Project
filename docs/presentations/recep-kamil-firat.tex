\begin{frame}[fragile]{GPR mimari}
\begin{lstlisting}
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel, WhiteKernel

kernel = (
    ConstantKernel(1.0, (1e-3, 1e3))
    * RBF(length_scale=1.0, length_scale_bounds=(1e-2, 1e3))
    + WhiteKernel(noise_level=1e-3, noise_level_bounds=(1e-6, 1e-1))
)

base = GaussianProcessRegressor(
    kernel=kernel, normalize_y=True,
    n_restarts_optimizer=5, random_state=42, alpha=1e-8,
)

model = TransformedTargetRegressor(
    regressor=base, transformer=StandardScaler(),
)
\end{lstlisting}

  \begin{itemize}\small
    \item Kernel hiperparametreleri marginal-likelihood ile fit
    \item \code{normalize\_y} \(+\) \code{TransformedTargetRegressor} \(\Rightarrow\) hedef ölçeği bağımsız
    \item Posterior kapalı form \(\Rightarrow\) her tahminde \(\sigma\) bedava
  \end{itemize}
\end{frame}

\begin{frame}{CV stratejisi: \textit{neden} grouped?}
  \textbf{Problem:} 270 satırın 108'i ANSYS re-run'ı --- aynı
  \((m, K, \theta)\) noktasında ufak sayısal farklarla tekrar eden satırlar.

  \vspace{0.3cm}
  \begin{itemize}
    \item Standart \code{KFold}\,/\,\code{LOO} kullanırsak: aynı grid noktası
          hem train hem test setine düşebilir \(\Rightarrow\) \alert{leakage}
    \item Görünür sonuç: \(R^2\) yapay olarak \(\approx 1.0\)'a çıkar
    \item Gerçek out-of-sample performansını ölçemeyiz
  \end{itemize}

  \vspace{0.4cm}
  \textbf{Çözüm:} \code{GroupKFold(n\_splits=5)}, group ID Ata'nın
  \code{grid\_groups()} fonksiyonundan geliyor:
  \begin{center}
    \texttt{"\,m\,|\,K\,|\,theta\,"} \quad \(\Rightarrow\) \quad re-run'lar aynı fold'da
  \end{center}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}[fragile]{Eğitim döngüsü}
\begin{lstlisting}
from sklearn.model_selection import GroupKFold

def _grouped_cv_eval(X, y, groups, n_splits):
    preds = np.full(len(y), np.nan)
    for tr, te in GroupKFold(n_splits).split(X, y, groups=groups):
        model, _ = _make_estimator()
        model.fit(X.iloc[tr], y.iloc[tr])
        preds[te] = model.predict(X.iloc[te])
    return {
        "r2_cv": r2_score(y, preds),
        "mae_cv": mean_absolute_error(y, preds),
    }
\end{lstlisting}

  \vspace{0.3cm}
  Her hedef için:
  \begin{enumerate}\small
    \item 5 fold üzerinde out-of-fold tahmin
    \item Tüm satırlar üzerinde final fit \(\Rightarrow\) \file{models/<target>.joblib}
    \item Metrikler \file{models/metrics.json}'a yazılır
  \end{enumerate}
\end{frame}

\begin{frame}{Dürüst limitler}
  \begin{itemize}
    \item \textbf{İnterpolasyon güvenilir, ekstrapolasyon değil.}
          \(m < 50\unit{kg}\), \(K > 6\), \(\theta > 20°\) için tahminler
          dikkatle kullanılmalı.
    \item \textbf{18-yüz fixed support sabit.} Farklı fiksasyon
          desenleri modellenmiyor; yeni FEA grid gerekir.
    \item \textbf{\(F_y = 0\) tüm satırlarda.} Düzlem dışı yükleme yok;
          gerçek aktivite çoğunlukla düzlem-içi olduğundan kabul edilebilir
          ama tam değil.
    \item \textbf{Kernel hiperparametreleri bazı hedeflerde sınıra yakın
          (ConvergenceWarning).} \(R^2\)'de marjinal etki, ileride bound'lar
          genişletilebilir.
  \end{itemize}
  \vspace{0.4cm}
  \footnotesize
  Fallback zinciri: \code{GPR \(\to\) XGBoost \(\to\) sklearn-GBR} ---
  sklearn'in GPR'si bir ortamda yoksa diğerleri devreye girer
  (\code{test\_make\_estimator\_falls\_back\_to\_sklearn\_gbr}).
\end{frame}
